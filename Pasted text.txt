        function handleLoginSubmit() {

            const phoneInput = document.getElementById('authPhone').value.trim();

            const passInput = document.getElementById('authPass').value.trim();



            if(!phoneInput || !passInput) {

                showToast("សូមបំពេញលេខទូរសព្ទ និង Password!");

                return;

            }



            if(isRegisterView) {

                const nameInput = document.getElementById('regName').value.trim();

                if(!nameInput) { showToast("សូមបំពេញ Username!"); return; }



                if(customersDB.some(c => c.phone === phoneInput)) {

                    showToast("លេខទូរសព្ទនេះត្រូវបានចុះឈ្មោះរួចហើយ!");

                    return;

                }



                const newAccount = { name: nameInput, phone: phoneInput, password: passInput, status: "active" };

                customersDB.push(newAccount);

                loggedInCustomer = newAccount;

                showToast("បង្កើតគណនីបានជោគជ័យ!");

                updateTopNavAuth();

                processPendingActionAfterLogin();

            } else {

                const found = customersDB.find(c => c.phone === phoneInput && c.password === passInput);

                if(found) {

                    if(found.status === 'blocked') {

                        showToast("គណនីរបស់អ្នកត្រូវបាន Block / បិទដំណើការ!");

                        return;

                    }

                    loggedInCustomer = found;

                    showToast("ចូលគណនីបានជោគជ័យ!");

                    updateTopNavAuth();

                    processPendingActionAfterLogin();

                } else {

                    showToast("ព័ត៌មានមិនត្រឹមត្រូវ!");

                }

            }

        }

        function filterAndSortProducts() {

            const query = document.getElementById('searchInput').value.toLowerCase();

            const sortVal = document.getElementById('sortSelect').value;

           

            let filtered = products.filter(p => p.name.toLowerCase().includes(query));



            if(sortVal === 'low-high') {

                filtered.sort((a,b) => a.price - b.price);

            } else if(sortVal === 'high-low') {

                filtered.sort((a,b) => b.price - a.price);

        /* PREVIEW & PROCESS 1 TO 6 PICTURES */

        function previewSelectedImages(event) {

            const files = Array.from(event.target.files).slice(0, 6);

            if(!files.length) return;



            currentFormImages = [];

            let loaded = 0;



            files.forEach((file, idx) => {

                const reader = new FileReader();

                reader.onload = function(e) {

                    currentFormImages[idx] = e.target.result;

                    loaded++;

                    if(loaded === files.length) {

                        updateImagePreviewGrid(currentFormImages);

                    }

                };

                reader.readAsDataURL(file);

            });

        }



        function updateImagePreviewGrid(images) {

            const container = document.getElementById('adminImgPreviewGrid');

            let html = '';

            for(let i = 0; i < 6; i++) {

                if(images[i]) {

                    html += `<div class="preview-box"><img src="${images[i]}"></div>`;

                } else {

                    html += `<div class="preview-box">Pic ${i + 1}</div>`;

                }

            }

            container.innerHTML = html;

        }
